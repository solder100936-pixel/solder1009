## 程式全文

程式放在 scratchpad（不放進倉庫），執行方式：`python3 -I check_events.py research/30_生活事件草稿/30_events_draft.json`。

檢查項目：
- (a) JSON 可解析；最外層是陣列；每則欄位齊全且順序固定（id、text、text2、stage、months、signal、tags、life、means、weight、freq_basis、src、sensitive、status）；型別正確；id 為 `E_` 加小寫英數與底線、不重複、不撞現有 31 則；stage 只用五個合法值；months 為 1～12 的整數；weight 為 1～5 的整數；status 全為 draft；means 以「你那時候看到的是」開頭；遊戲文字欄位的佔位符只用 {name}、{ta}、{cg}、{oth}。
- (b) 掃描 text、text2、life、means 的禁字：應該、記得要、提醒、建議、其實這是、正確、錯誤。

```python
"""驗證 30_events_draft.json 並掃描禁字。用法：python3 -I check_events.py <json 路徑>"""
import json, re, sys

EXISTING = {'E_bags', 'E_corner', 'E_letter', 'E_bbq', 'E_newyear', 'E_choke', 'E_pillbox', 'E_shoes',
            'E_altar', 'E_flowers', 'E_tv', 'E_chat', 'E_rouwan', 'E_exam', 'E_bath', 'E_cg_day_off',
            'E_cg_call', 'E_cg_note', 'E_cg_doctor', 'E_cg_night', 'E_ltc_noone', 'E_sib_money',
            'E_ward_chair', 'E_good_news', 'E_alarm', 'E_outage_notice', 'E_typhoon', 'E_rt_home',
            'E_ricecake', 'E_hands', 'E_calm_walk'}
FIELDS = [('id', str), ('text', str), ('text2', str), ('stage', list), ('months', list), ('signal', str),
          ('tags', list), ('life', str), ('means', str), ('weight', int), ('freq_basis', str), ('src', str),
          ('sensitive', bool), ('status', str)]
STAGES = {'還能自己走動', '開始需要人扶', '住院中', '在家用呼吸器', '在機構'}
BANNED = ['應該', '記得要', '提醒', '建議', '其實這是', '正確', '錯誤']
SCAN = ['text', 'text2', 'life', 'means']
PLACEHOLDER_OK = {'{name}', '{ta}', '{cg}', '{oth}'}

path = sys.argv[1]
errs, hits = [], []
with open(path, encoding='utf-8') as f:
    data = json.load(f)  # (a) 可解析
print(f'JSON 可解析：{len(data)} 則')
if not isinstance(data, list):
    errs.append('最外層不是陣列')
seen = set()
for i, e in enumerate(data):
    tag = f'#{i} {e.get("id", "?")}'
    keys = list(e.keys())
    if keys != [k for k, _ in FIELDS]:
        errs.append(f'{tag} 欄位或順序不符：{keys}')
    for k, t in FIELDS:
        if k not in e:
            continue
        v = e[k]
        if t is int and isinstance(v, bool) or not isinstance(v, t):
            errs.append(f'{tag} {k} 型別錯：{type(v).__name__}')
    eid = e.get('id', '')
    if not re.fullmatch(r'E_[a-z0-9_]+', eid):
        errs.append(f'{tag} id 格式不符')
    if eid in seen:
        errs.append(f'{tag} id 重複')
    if eid in EXISTING:
        errs.append(f'{tag} id 與現有 31 則相同')
    seen.add(eid)
    st = e.get('stage', [])
    if not st or any(s not in STAGES for s in st):
        errs.append(f'{tag} stage 值不合法：{st}')
    if any(not (isinstance(m, int) and not isinstance(m, bool) and 1 <= m <= 12) for m in e.get('months', [])):
        errs.append(f'{tag} months 值不合法')
    if any(not isinstance(x, str) or not x for x in e.get('tags', [])) or not e.get('tags'):
        errs.append(f'{tag} tags 不合法')
    w = e.get('weight')
    if not (isinstance(w, int) and not isinstance(w, bool) and 1 <= w <= 5):
        errs.append(f'{tag} weight 不在 1～5')
    if e.get('status') != 'draft':
        errs.append(f'{tag} status 不是 draft')
    for k in ('text', 'life', 'means', 'signal', 'freq_basis', 'src'):
        if not e.get(k):
            errs.append(f'{tag} {k} 空白')
    if not e.get('means', '').startswith('你那時候看到的是'):
        errs.append(f'{tag} means 開頭不是「你那時候看到的是」')
    for k in SCAN:
        for ph in re.findall(r'\{[^}]*\}', e.get(k, '')):
            if ph not in PLACEHOLDER_OK:
                errs.append(f'{tag} {k} 有未知佔位符 {ph}')
    # (b) 禁字掃描
    for k in SCAN:
        for b in BANNED:
            if b in e.get(k, ''):
                hits.append((eid, k, b, e[k]))

print(f'驗證錯誤：{len(errs)} 筆')
for x in errs:
    print('  ', x)
print(f'禁字命中：{len(hits)} 筆（掃描欄位 {"、".join(SCAN)}；禁字 {"、".join(BANNED)}）')
for h in hits:
    print('  ', h)
sys.exit(1 if errs or hits else 0)
```
